86inline std::uint64_t ParseUnsignedText(const std::string& text,
87 const std::string& field_name) {
88 const std::string normalized = TrimCopy(text);
89 if (normalized.empty()) {
90 throw std::invalid_argument("字段 " + field_name + " 不能为空");
91 }
92 if (normalized[0] == '-') {
93 throw std::invalid_argument("字段 " + field_name + " 不能为负数");
94 }
95
96 errno = 0;
97 std::size_t consumed = 0;
98 const unsigned long long parsed = std::stoull(normalized, &consumed, 10);
99 if (errno == ERANGE || consumed != normalized.size()) {
100 throw std::invalid_argument("字段 " + field_name + " 必须是十进制整数");
101 }
102 return static_cast<std::uint64_t>(parsed);
103}

237inline void ParseAddressText(const std::string& text,
238 std::vector<std::uint8_t>* addresses,
239 std::set<int>* seen) {
240 std::string normalized = TrimCopy(text);
241 if (normalized.empty()) {
242 throw std::invalid_argument("addresses 不能为空");
243 }
244
245 std::size_t begin = 0;
246 while (begin < normalized.size()) {
247 const std::size_t comma = normalized.find(',', begin);
248 const std::string item = TrimCopy(normalized.substr(
249 begin, comma == std::string::npos ? std::string::npos : comma - begin));
250 if (item.empty()) {
251 throw std::invalid_argument("addresses 存在空地址项");
252 }
253
254 const std::size_t dash = item.find('-');
255 if (dash == std::string::npos) {
256 const std::uint64_t address = ParseUnsignedText(item, "addresses");
257 if (address > 255) {
258 throw std::invalid_argument("设备地址必须在 1～255 之间");
259 }
260 AppendAddress(static_cast<int>(address), addresses, seen);
261 } else {
262 if (item.find('-', dash + 1) != std::string::npos) {
263 throw std::invalid_argument("addresses 范围格式错误");
264 }
265 const std::uint64_t start = ParseUnsignedText(
266 TrimCopy(item.substr(0, dash)), "addresses");
267 const std::uint64_t end = ParseUnsignedText(
268 TrimCopy(item.substr(dash + 1)), "addresses");
269 if (start > 255 || end > 255) {
270 throw std::invalid_argument("设备地址必须在 1～255 之间");
271 }
272 if (start > end) {
273 throw std::invalid_argument("addresses 范围的起始地址不能大于结束地址");
274 }
275 for (std::uint64_t address = start; address <= end; ++address) {
276 AppendAddress(static_cast<int>(address), addresses, seen);
277 }
278 }
279
280 if (comma == std::string::npos) {
281 break;
282 }
283 begin = comma + 1;
284 }

287inline std::vector<std::uint8_t> ParseAddresses(const nlohmann::json& value) {
288 std::vector<std::uint8_t> addresses;
289 std::set<int> seen;
290
291 if (value.is_string()) {
292 ParseAddressText(value.get<std::string>(), &addresses, &seen);
293 } else if (value.is_array()) {
294 for (nlohmann::json::const_iterator it = value.begin(); it != value.end(); ++it) {
295 if (it->is_string()) {
296 ParseAddressText(it->get<std::string>(), &addresses, &seen);
297 } else {
298 const std::uint64_t address = JsonUnsigned(*it, "addresses", 0);
299 if (address > 255) {
300 throw std::invalid_argument("设备地址必须在 1～255 之间");
301 }
302 AppendAddress(static_cast<int>(address), &addresses, &seen);
303 }
304 }
305 } else {
306 const std::uint64_t address = JsonUnsigned(value, "addresses", 0);
307 if (address > 255) {
308 throw std::invalid_argument("设备地址必须在 1～255 之间");
309 }
310 AppendAddress(static_cast<int>(address), &addresses, &seen);
311 }
312
313 if (addresses.empty()) {
314 throw std::invalid_argument("addresses 不能为空");
315 }
316 return addresses;
317}

336inline std::string TimestampNow() {
337 const std::chrono::system_clock::time_point now = std::chrono::system_clock::now();
338 const std::chrono::milliseconds milliseconds =
339 std::chrono::duration_cast<std::chrono::milliseconds>(now.time_since_epoch());
340 const std::time_t time_value = std::chrono::system_clock::to_time_t(now);
341
342 std::tm local_time;
343#if defined(_WIN32)
344 localtime_s(&local_time, &time_value);
345#else
346 localtime_r(&time_value, &local_time);
347#endif
348
349 char date_buffer[32] = {0};
350 std::strftime(date_buffer, sizeof(date_buffer), "%Y-%m-%d %H:%M:%S", &local_time);
351 const long millisecond_part = static_cast<long>(milliseconds.count() % 1000);
352
353 std::ostringstream stream;
354 stream << date_buffer << '.' << std::setfill('0') << std::setw(3) << millisecond_part;
355 return stream.str();
356}
