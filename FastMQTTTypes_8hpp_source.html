47inline const char* LifecycleStateToString(LifecycleState s) {

52 case LifecycleState::Running: return "Running";

112 static FastMQTTConfig FromJson(const nlohmann::json& j) {
113 // 兼容顶层包含 "mqtt" 的情况。
114 const nlohmann::json& m = (j.contains("mqtt") && j["mqtt"].is_object())
115 ? j["mqtt"]
116 : j;
117 FastMQTTConfig cfg;
118 cfg.enable = m.value("enable", true);
119
120 if (m.contains("broker") && m["broker"].is_object()) {
121 const auto& b = m["broker"];
122 cfg.broker.host = b.value("host", cfg.broker.host);
123 cfg.broker.port = b.value("port", cfg.broker.port);
124 cfg.broker.client_id = b.value("client_id", cfg.broker.client_id);
125 cfg.broker.username = b.value("username", cfg.broker.username);
126 cfg.broker.password = b.value("password", cfg.broker.password);
127 cfg.broker.keep_alive = b.value("keep_alive", cfg.broker.keep_alive);
128 cfg.broker.clean_session = b.value("clean_session", cfg.broker.clean_session);
129 cfg.broker.auto_reconnect = b.value("auto_reconnect", cfg.broker.auto_reconnect);
130 cfg.broker.connect_timeout_ms = b.value("connect_timeout_ms", cfg.broker.connect_timeout_ms);
131 }
132
133 if (m.contains("thread") && m["thread"].is_object()) {
134 const auto& t = m["thread"];
135 cfg.thread.send_queue_size = t.value("send_queue_size", cfg.thread.send_queue_size);
136 cfg.thread.recv_queue_size = t.value("recv_queue_size", cfg.thread.recv_queue_size);
137 }
138
139 if (m.contains("default") && m["default"].is_object()) {
140 const auto& d = m["default"];
141 cfg.def.qos = d.value("qos", cfg.def.qos);
142 cfg.def.retain = d.value("retain", cfg.def.retain);
143 }
144 return cfg;
145 }
146};
