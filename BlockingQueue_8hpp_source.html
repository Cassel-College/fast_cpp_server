3// =============================================================================
4// 文件：BlockingQueue.hpp
5// 模块：FastMQTT
6// 说明：线程安全的阻塞队列（Blocking Queue）。
7//
8// 设计要点：
9// 1. 使用 std::mutex + std::condition_variable 实现，禁止 Busy Loop（忙等待）。
10// 2. 支持最大容量限制，超过容量时 push 直接失败，避免内存无限增长（OOM）。
11// 3. 支持超时出队 pop(timeout)，方便后台线程周期性检查退出标志。
12// 4. 支持 shutdown()，唤醒所有阻塞在 pop 上的线程，用于安全退出。
13//
14// 该组件不依赖任何业务类型，可被 SendQueue / ReceiveQueue 复用。
15// =============================================================================

49 bool Push(const T& value) {
50 std::unique_lock<std::mutex> lock(mutex_);
51 if (shutdown_) {
52 return false;
53 }
54 // 容量限制：max_size_ 为 0 表示不限容量。
55 if (max_size_ != 0 && queue_.size() >= max_size_) {
56 return false; // 队列已满，直接拒绝，交由上层记录日志。
57 }
58 queue_.push_back(value);
59 // 只唤醒一个等待线程即可，避免惊群。
60 not_empty_.notify_one();
61 return true;
62 }
63
69 bool Push(T&& value) {
70 std::unique_lock<std::mutex> lock(mutex_);
71 if (shutdown_) {
72 return false;
73 }
74 if (max_size_ != 0 && queue_.size() >= max_size_) {
75 return false;
76 }
77 queue_.push_back(std::move(value));
78 not_empty_.notify_one();
79 return true;
80 }
81
87 bool Pop(T& out) {
88 std::unique_lock<std::mutex> lock(mutex_);
89 // 使用条件变量等待，条件：队列非空 或 已 shutdown。
90 not_empty_.wait(lock, [this] { return !queue_.empty() || shutdown_; });
91 if (queue_.empty()) {
92 // 只有在 shutdown 且队列已空时才会走到这里。
93 return false;
94 }
95 out = std::move(queue_.front());
96 queue_.pop_front();
97 return true;
98 }
99
106 bool Pop(T& out, int timeout_ms) {
107 std::unique_lock<std::mutex> lock(mutex_);
108 const bool ready = not_empty_.wait_for(
109 lock, std::chrono::milliseconds(timeout_ms),
110 [this] { return !queue_.empty() || shutdown_; });
111 if (!ready || queue_.empty()) {
112 return false; // 超时，或 shutdown 后队列为空。
113 }
114 out = std::move(queue_.front());
115 queue_.pop_front();
116 return true;
117 }
118
124 void Shutdown() {
125 std::unique_lock<std::mutex> lock(mutex_);
126 shutdown_ = true;
127 not_empty_.notify_all();
128 }
129
133 void Clear() {
134 std::unique_lock<std::mutex> lock(mutex_);
135 queue_.clear();
136 }

142 std::size_t Size() const {
143 std::unique_lock<std::mutex> lock(mutex_);
144 return queue_.size();
145 }
146
151 bool Empty() const {
152 std::unique_lock<std::mutex> lock(mutex_);
153 return queue_.empty();
154 }
155
159 bool IsShutdown() const {
160 std::unique_lock<std::mutex> lock(mutex_);
161 return shutdown_;
162 }
163
164private:
165 mutable std::mutex mutex_;
